## ksm

```cpp
int ksm(int a, int n) {
    a %= p;
    int res = 1;
    while(n){
        if(n & 1) res = a * res % p;
        a = a * a % p;
        n >>= 1;
    }
    return res % p;
}
```

## 矩阵快速幂
复杂度：O(k^3 * log(k))

什么时候使用：状态少，转移次数多

一个经典应用：

假设有一张图，它的邻接矩阵为 $A$。如果存在一条边：$u \to v$ 那么令：$A_{u,v}=1$，否则：$A_{u,v}=0$
邻接矩阵有一个非常重要的性质：
$(A^k)_{u,v}$表示：

> 从节点 $u$ 出发，**恰好走 $k$ 条边**到达节点 $v$ 的方案数。

所以，如果题目要求：

> 给定一张图，求从 $s$ 到 $t$ 恰好走 $k$ 条边的方案数，其中 $k$ 非常大。

就可以构造邻接矩阵 $A$，然后使用矩阵快速幂求：$A^k$
最终答案就是：$(A^k)_{s,t}$

```cpp
struct Matrix {
	int n, m;
	vector<vector<int>> a;
	Matrix(int n_, int m_) {
		n = n_;
		m = m_;
		a.assign(n, vector<int>(m));
	}
	vector<int>& operator[](int x) {
		return a[x]; // 可以像数组一样访问
	}

	Matrix operator*(Matrix &b) {
		Matrix c(n, b.m);
		for (int i = 0; i < n; i++) {
			for (int k = 0; k < m; k++) {
				for (int j = 0; j < b.m; j++) {
					c[i][j] = (c[i][j] + a[i][k] * b[k][j]) % p;
				}
			}
		}
		return c;
	}
};

Matrix ksm(Matrix a, int n) {
	Matrix res(a.n, a.n);
	for (int i = 0; i < a.n; i++) {
		res[i][i] = 1;
	}
	while(n) {
		if(n & 1) res = res * a;
		a = a * a;
		n >>= 1;
	}
	return res;
}
```

```cpp
int spf[N], primes[N], cnt;

void init() {
	for (int i = 2; i < N; i++) {
		if(!spf[i]) {
			spf[i] = i;
			primes[++cnt] = i;
		}

		for (int j = 1; j <= cnt && i * primes[j] < N; j++) {
			int p = primes[j];
			spf[i * p] = p;
			if(i % p == 0) break;

bool is_prime(int x) {

vector<int> get_prime_factor(int x) {
	vector<int> res;

	while(x > 1) {
		int p = spf[x];
		res.push_back(p);

		while(x % p == 0)
			x /= p;
	}

	return res;
}
```

```cpp
	if(!b) {
		x = 1;
		y = 0;
		return a;
	}
	int d = exgcd(b, a % b, y, x);
	y -= a / b * x;
	return d;
```

## 等差数列

### 一、等差数列 $\{a_n\}$
首项：$a_1$，公差：$d$

1. 通项公式
$$a_n = a_1 + (n-1)d$$

2. 前 $n$ 项和
$$S_n = na_1 + \frac{n(n-1)}{2}d$$

3. 常用性质
若 $m+n=p+q$，则
$$a_m+a_n=a_p+a_q$$

### 二、等比数列 $\{a_n\}$
首项：$a_1$，公比：$q\quad(q\neq0)$

1. 通项公式
$$a_n = a_1 q^{n-1}$$

2. 前 $n$ 项和
$q\neq1$：
$$S_n=\frac{a_1(1-q^n)}{1-q}=\frac{a_1-a_n q}{1-q}$$
$q=1$：
$$S_n = n a_1$$

3. 常用性质
若 $m+n=p+q$，则
$$a_m\cdot a_n=a_p\cdot a_q$$

## 线性基
异或线性基

可实现操作：1.将所有异或结果分解为基底，2.求异或和的最大最小值，3.对不同线性基集合进行& 和 | (求交，合并)
```cpp
template<class T>
struct Basis {
    constexpr static int K = 30;
    std::vector<T> e;
    
    Basis() {
        e.assign(K, 0);
    }

    bool add(T x) {
        for (int i = K - 1; i >= 0 && x; i--) {
            if (x >> i & 1) {
                if (e[i]) {
                    x ^= e[i];
                } else {
                    e[i] = x;
                    return true;
                }
            }
        }
        return false;
    }

    T max(T x = 0) {
        for (int i = K - 1; i >= 0; i--) {
            x = std::max(x, x ^ e[i]);
        }
        return x;
    }
    T min(T x) {
        for (int i = K - 1; i >= 0; i--) {
            x = std::min(x, x ^ e[i]);
        }
        return x;
    }

    void work() {
        for (int i = 0; i < K; i++) {
            for (int j = i + 1; j < K; j++) {
                if (e[j] >> i & 1) {
                    e[j] ^= e[i];
                }
            }
        }
    }

    constexpr Basis &operator|=(const Basis &rhs) & {
        for (int i = 0; i < rhs.K; i++) {
            add(rhs.e[i]);
        }
        return *this;
    }
    constexpr Basis &operator&=(const Basis &rhs) & {
        Basis ans;
        std::vector<T> b(rhs.K);
        for (int i = rhs.K - 1; i >= 0; i--) {
            T x = rhs.e[i], v = x;
            for (int k = i; k >= 0 && x; k--) {
                if (x >> k & 1) {
                    if (!e[k]) {
                        e[k] = x;
                        b[k] = v;
                    }
                    x ^= e[k];
                    v ^= b[k];
                }
            }
            ans.add(v);
        }

        *this = std::move(ans);
        return *this;
    }

    constexpr friend Basis operator|(Basis lhs, const Basis &rhs) {
        lhs |= rhs;
        return lhs;
    }
    constexpr friend Basis operator&(Basis lhs, const Basis &rhs) {
        lhs &= rhs;
        return lhs;
    }
};
```

## 二项式定理

1. 定义

二项式定理用于展开二项式的 n 次方：

\[
(a+b)^n=\sum_{k=0}^{n}C_n^k a^{n-k}b^k
\]

其中：

\[
C_n^k=\binom nk=\frac{n!}{k!(n-k)!}
\]

含义：
从 n 个括号中选择 k 个取 b，其余取 a，因此系数为组合数。

---


2. 杨辉三角

组合数满足：

\[
C_n^k=C_{n-1}^{k-1}+C_{n-1}^{k}
\]

杨辉三角第 n 行：

\[
C_n^0,C_n^1,...,C_n^n
\]

---

3. 常用性质

(1) 系数和

令 a=b=1：

\[
(1+1)^n=2^n
\]

所以：

\[
\sum_{k=0}^{n}C_n^k=2^n
\]


(2) 交错和

令 a=1,b=-1：

\[
(1-1)^n=0
\]

所以：

\[
C_n^0-C_n^1+C_n^2-\cdots=0
\]

## 高斯消元
拓展：差分是可逆的线性变换，不改变矩阵的线性关系（秩）

### 普通线性方程
ans 为答案数组，ans里面需要判断和 eps 的大小，小于eps的输出0

0 无解，1唯一解，2无穷多解
```cpp
const long double eps = 1e-12;
int gauss(vector<vector<long double>> a, vector<long double> &ans) {
	int n = a.size();
	int m = a[0].size() - 1;
	vector<int> pos(m, -1);
	int row = 0;
	for (int col = 0; col < m && row < n; col++) {
		int mx = row;
		for (int i = row; i < n; i++) {
			if(fabs(a[i][col]) > fabs(a[mx][col])) {
				mx = i;
			}
		}

		if(fabs(a[mx][col]) < eps) continue;
		
		swap(a[mx], a[row]);
		pos[col] = row;
		for (int i = 0; i < n; i++) {
			if(i == row) continue;
			long double t = a[i][col] / a[row][col];
			for (int j = col; j <= m; j++) {
				a[i][j] -= t * a[row][j];
			}
		}
		row++;
        // row为方程的秩
        // if(row == min(n, m)) {
        //     满秩
        // }
	}

	for (int i = 0; i < n; i++) {
		bool zero = true;
		for (int j = 0; j < m; j++) {
			if(fabs(a[i][j]) > eps) {
				zero = false;
				break;
			}
		}

		if(zero && fabs(a[i][m]) > eps) {
			return 0;
		}
	}
	ans.assign(m, 0);

	for (int i = 0; i < m; i++) {
		if(pos[i] != -1) {
			ans[i] = a[pos[i]][m] / a[pos[i]][i];
		}
	}

	for (int i = 0; i < m; i++) {
		if(pos[i] == -1) {
			return 2;
		}
	}

	return 1;
}
```

### 异或方程
```cpp
const int M = 2005;

int gauss(vector<bitset<M>> a, int m, vector<int> &ans) {
	int n = a.size();
	vector<int> pos(m, -1);
	int row = 0;
	for (int col = 0; col < m && row < n; col++) {
		int p = row;
		while(p < n && !a[p][col]) {
			p++;
		}
		if(p == n) {
			continue;
		}
		swap(a[p], a[row]);
		pos[col] = row;

		for (int i = 0; i < n; i++) {
			if(i != row && a[i][col]) {
				a[i] ^= a[row];
			}
		}
		row++;
	}

	for (int i = 0; i < n; i++) {
		bool zero = true;
		for (int j = 0; j < m; j++) {
			if(a[i][j]) {
				zero = false;
				break;
			}
		}
		if(zero && a[i][m]) {
			return 0;
		}
	}
	ans.assign(m, 0);

	for (int i = 0; i < m; i++) {
		if(pos[i] != -1) {
			ans[i] = a[pos[i]][m];
		}
	}
	for (int i = 0; i < m; i++) {
		if(pos[i] == -1) {
			return 2;
		}
	}
	return 1;
}
```

## NTT/FFT

### NTT
数组 0 位存的是常数项
```cpp
const int g = 3;

int ksm(int a, int n) {
	a %= p;
	int res = 1;
	while(n) {
		if(n & 1) {
			res = res * a % p;
		}
		a = a * a % p;
		n >>= 1;
	}
	return res;
}

void ntt(vector<int> &a, int op) {
	int n = a.size();

	for(int i = 1, j = 0; i < n; i++) {
		int bit = n >> 1;
		for(; j & bit; bit >>= 1) {
			j ^= bit;
		}
		j ^= bit;
		if(i < j) {
			swap(a[i], a[j]);
		}
	}

	for(int len = 2; len <= n; len <<= 1) {
		int wlen = ksm(g, (p - 1) / len);
		if(op == -1) {
			wlen = ksm(wlen, p - 2);
		}

		for(int i = 0; i < n; i += len) {
			int w = 1;
			for(int j = 0; j < len / 2; j++) {
				int u = a[i + j];
				int v = a[i + j + len / 2] * w % p;
				a[i + j] = (u + v) % p;
				a[i + j + len / 2] = (u - v + p) % p;
				w = w * wlen % p;
			}
		}
	}

	if(op == -1) {
		int inv = ksm(n, p - 2);
		for(auto &x : a) {
			x = x * inv % p;
		}
	}
}

vector<int> mul(vector<int> a, vector<int> b) {
	int sz = a.size() + b.size() - 1;
	int n = 1;
	while(n < sz) {
		n <<= 1;
	}
	a.resize(n);
	b.resize(n);

	ntt(a, 1);
	ntt(b, 1);
	for(int i = 0; i < n; i++) {
		a[i] = a[i] * b[i] % p;
	}
	ntt(a, -1);
	a.resize(sz);

	return a;
}
```

### FFT
```cpp
const double pi = acos(-1);
using cd = complex<double>;

void fft(vector<cd> &a, int op) {
	int n = a.size();
	for(int i = 1, j = 0; i < n; i++) {
		int bit = n >> 1;
		for(; j & bit; bit >>= 1) {
			j ^= bit;
		}
		j ^= bit;
		if(i < j) {
			swap(a[i], a[j]);
		}
	}

	for(int len = 2; len <= n; len <<= 1) {
		double ang = 2 * pi / len * op;
		cd wlen(cos(ang), sin(ang));
		for(int i = 0; i < n; i += len) {
			cd w(1);
			for(int j = 0; j < len / 2; j++) {
				cd u = a[i + j];
				cd v = a[i + j + len / 2] * w;
				a[i + j] = u + v;
				a[i + j + len / 2] = u - v;
				w *= wlen;
			}
		}
	}

	if(op == -1) {
		for(auto &x : a) {
			x /= n;
		}
	}
}

	int n = 1;

		n <<= 1;
	}

	vector<cd> fa(n), fb(n);
	for(int i = 0; i < a.size(); i++) {
		fa[i] = a[i];
	}
	for(int i = 0; i < b.size(); i++) {
		fb[i] = b[i];
	}
	fft(fa, 1);
	fft(fb, 1);
	for(int i = 0; i < n; i++) {
		fa[i] *= fb[i];
	}
	fft(fa, -1);

	vector<int> c(sz);
	for(int i = 0; i < sz; i++) {
		c[i] = (long long)(fa[i].real() + 0.5);
	}
	return c;
}
```
