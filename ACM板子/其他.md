## 对拍
数据生成器
```cpp
#include <bits/stdc++.h>
using namespace std;
mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
int rnd(int l, int r) {
	return uniform_int_distribution<int>(l, r)(rng);
}

int main() {
	int n = rnd(1, 10);
	cout << n << endl;
	for (int i = 1; i <= n; i++) {
		cout << rnd(1, 20) << " ";
	}
	cout << endl;
	return 0;
}
```
对拍程序
```cpp
#include<bits/stdc++.h>
using namespace std;

int main() {
	system("g++ -std=c++11 gen.cpp -o gen.exe");
	system("g++ -std=c++11 std.cpp -o std.exe");
	system("g++ -std=c++11 brute.cpp -o brute.exe");
	
	int cnt = 0;
	while(1) {
		cnt++;
		system("gen.exe > data.txt");
		system("brute.exe < data.txt > brute.txt");
		system("std.exe < data.txt > std.txt");
		
		if(system("fc std.txt brute.txt")) {
			cout << "WA" << endl;
			system("type data.txt");
			system("type brute.txt");
			system("type std.txt");
			break; 
		}
		else cout << cnt << " AC" << endl;
	}
}
```

```cpp
    auto ternary = [&](int l, int r) -> int {
        while(r - l > 3) {
            int m1 = l + (r - l) / 3;
            int m2 = r - (r - l) / 3;

            if(check(m1) < check(m2)) l = m1 + 1; // 最大值
```

## 高精度
减法必须 a > b
```cpp
struct BigInt {
	vector<int> a;
	const int base = 1e9;

	BigInt(string s = "0") {
		for(int i = s.size(); i > 0; i -= 9) {
			int l = max(0ll, i - 9);
			a.push_back(stoi(s.substr(l, i - l)));
		}
		while(a.size() > 1 && a.back() == 0)
			a.pop_back();
	}

	BigInt operator+(const BigInt& b) {
		BigInt c;
		c.a.clear();

		int t = 0;
		for(int i = 0; i < max(a.size(), b.a.size()) || t; i++) {
			int cur = t;
			if(i < a.size()) cur += a[i];
			if(i < b.a.size()) cur += b.a[i];

			c.a.push_back(cur % base);
			t = cur / base;
		}

		return c;
	}

	BigInt operator-(const BigInt& b) {
		BigInt c;
		c.a.clear();

		int t = 0;

		for(int i = 0; i < a.size(); i++) {
			int cur = a[i] - t;

			if(i < b.a.size())
				cur -= b.a[i];

			if(cur < 0) {
				cur += base;
				t = 1;
			}
			else {
				t = 0;
			}

			c.a.push_back(cur);
		}

		while(c.a.size() > 1 && c.a.back() == 0)
			c.a.pop_back();

		return c;
	}

	BigInt operator*(int b) {
		BigInt c;
		c.a.clear();

		long long t = 0;

		for(auto x : a) {
			long long cur = 1ll * x * b + t;
			c.a.push_back(cur % base);
			t = cur / base;
		}

		while(t) {
			c.a.push_back(t % base);
			t /= base;
		}

		return c;
	}

	BigInt operator/(int b) {
		BigInt c;
		c.a.assign(a.size(), 0);

		long long t = 0;

		for(int i = a.size() - 1; i >= 0; i--) {
			long long cur = a[i] + t * base;
			c.a[i] = cur / b;
			t = cur % b;
		}

		while(c.a.size() > 1 && c.a.back() == 0)
			c.a.pop_back();

		return c;
	}

	void print() {
		cout << a.back();

		for(int i = a.size() - 2; i >= 0; i--) {
			cout << setw(9) << setfill('0') << a[i];
		}

		cout << endl;
	}
};
```
