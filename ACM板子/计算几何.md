## 基础点线
```cpp
using ld = long double;
const ld eps = 1e-12;
const ld PI = acosl(-1);

// 判断浮点数符号
int sgn(ld x) {
	if(fabsl(x) < eps) return 0;
	return x > 0 ? 1 : -1;
}

struct Point {
	ld x, y;
	Point(ld x_ = 0, ld y_ = 0) : x(x_), y(y_) {} // 构造点
	Point operator + (const Point &p) const { return {x + p.x, y + p.y}; }
	Point operator - (const Point &p) const { return {x - p.x, y - p.y}; }
	Point operator * (ld k) const { return {x * k, y * k}; }
	Point operator / (ld k) const { return {x / k, y / k}; }
	Point operator - () const { return {-x, -y}; } // 向量取反
	bool operator == (const Point &p) const { return !sgn(x - p.x) && !sgn(y - p.y); }// 判断两点是否相等
	bool operator != (const Point &p) const { return !(*this == p); }
	bool operator < (const Point &p) const {
		if(x != p.x) return x < p.x;
		return y < p.y;
	}// 按 x 第一关键字、y 第二关键字排序
};

struct Line {
	Point a, b;
	Line(Point a_ = {}, Point b_ = {}) : a(a_), b(b_) {} // 构造直线或线段
};

ld dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
ld cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
ld cross(Point a, Point b, Point c) { return cross(b - a, c - a); }
/*
求向量 ab 与 ac 的叉积
> 0    c 在 ab 左边，逆时针
< 0    c 在 ab 右边，顺时针
= 0    a,b,c 共线
*/

ld len2(Point p) { return dot(p, p); } // 求向量长度平方
ld len(Point p) { return sqrtl(len2(p)); } // 求向量长度
ld dist(Point a, Point b) { return len(a - b); } // 两点距离
Point normalize(Point p) { return p / len(p); } // 求非0向量的单位向量
Point rotate90(Point p) { return {-p.y, p.x}; } // 向量逆时针旋转 90 度

// 向量逆时针旋转 rad 弧度
Point rotate(Point p, ld rad) {
	return {p.x * cosl(rad) - p.y * sinl(rad), p.x * sinl(rad) + p.y * cosl(rad)};
}

// 求两个向量的夹角
ld angle(Point a, Point b) {
	ld x = dot(a, b) / len(a) / len(b);
	x = max((ld)-1, min((ld)1, x));
	return acosl(x);
}

// 判断两直线是否平行
bool parallel(Line l1, Line l2) { return !sgn(cross(l1.b - l1.a, l2.b - l2.a)); }

// 判断点是否在有向直线左侧
bool pointOnLineLeft(Point p, Line l) { return sgn(cross(l.b - l.a, p - l.a)) > 0; }

// 判断点 p 是否在线段 ab 上
bool onSegment(Point p, Point a, Point b) {
	return !sgn(cross(a, b, p)) && sgn(dot(p - a, p - b)) <= 0;
}

// 判断点 p 是否在线段 l 上
bool onSegment(Point p, Line l) { return onSegment(p, l.a, l.b); }

// 求两条不平行直线交点
Point lineIntersection(Line l1, Line l2) {
	Point v = l1.b - l1.a, w = l2.b - l2.a;
	ld t = cross(l2.a - l1.a, w) / cross(v, w);
	return l1.a + v * t;
}

// 求点 p 在直线 l 上的投影点
Point projection(Point p, Line l) {
	Point v = l.b - l.a;
	return l.a + v * (dot(p - l.a, v) / dot(v, v));
}

// 求点 p 关于直线 l 的对称点
Point reflection(Point p, Line l) {
	Point h = projection(p, l);
	return h * 2 - p;
}

// 求点 p 到直线 l 的距离
ld distPointLine(Point p, Line l) {
	return fabsl(cross(l.b - l.a, p - l.a)) / len(l.b - l.a);
}

// 求点 p 到线段 l 的距离
ld distPointSegment(Point p, Line l) {
	if(l.a == l.b) return dist(p, l.a);
	if(sgn(dot(p - l.a, l.b - l.a)) < 0) return dist(p, l.a);
	if(sgn(dot(p - l.b, l.a - l.b)) < 0) return dist(p, l.b);
	return distPointLine(p, l);
}

// 判断两线段是否严格相交
bool properIntersect(Point a, Point b, Point c, Point d) {
	return sgn(cross(a, b, c)) * sgn(cross(a, b, d)) < 0 &&
		   sgn(cross(c, d, a)) * sgn(cross(c, d, b)) < 0;
}

// 判断线段 ab 与 cd 是否相交，包括端点和共线重叠
bool segmentIntersect(Point a, Point b, Point c, Point d) {
	int c1 = sgn(cross(a, b, c)), c2 = sgn(cross(a, b, d));
	int c3 = sgn(cross(c, d, a)), c4 = sgn(cross(c, d, b));
	if(c1 * c2 < 0 && c3 * c4 < 0) return true;
	if(c1 == 0 && onSegment(c, a, b)) return true;
	if(c2 == 0 && onSegment(d, a, b)) return true;
	if(c3 == 0 && onSegment(a, c, d)) return true;
	if(c4 == 0 && onSegment(b, c, d)) return true;
	return false;
}

// 判断两线段是否相交
bool segmentIntersect(Line l1, Line l2) {
	return segmentIntersect(l1.a, l1.b, l2.a, l2.b);
}

// 求两线段之间的最短距离
ld distSegmentSegment(Line l1, Line l2) {
	if(segmentIntersect(l1, l2)) return 0;
	return min({distPointSegment(l1.a, l2), distPointSegment(l1.b, l2), distPointSegment(l2.a, l1), distPointSegment(l2.b, l1)});
}

// 判断向量位于极角排序的哪个半平面
int half(Point p) {
	if(sgn(p.y)) return p.y < 0;
	return p.x < 0;
}

// 极角排序比较函数，从 x 轴正方向开始逆时针排序
bool polarCmp(Point a, Point b) {
    int ha = half(a), hb = half(b);
    if(ha != hb) return ha < hb;
    int c = sgn(cross(a, b));
    if(c) return c > 0;
    return len2(a) < len2(b);
}
```

## 多边形
```cpp
// 求多边形面积
ld polygonArea(const vector<Point> &a) {
	int n = a.size();
	ld res = 0;
	for (int i = 0; i < n; i++) res += cross(a[i], a[(i + 1) % n]);
	return fabs(res) / 2;
}

// 求多边形周长
ld perimeter(const vector<Point> &a) {
	int n = a.size();
	ld res = 0;
	for (int i = 0; i < n; i++) res += dist(a[i], a[(i + 1) % n]);
	return res;
}

// 判断点与多边形位置：1 内部，0 边界，-1 外部
int pointInPolygon(Point p, const vector<Point> &a) {
	int n = a.size();
	bool in = false;
	for (int i = 0; i < n; i++) {
		Point x = a[i], y = a[(i + 1) % n];
		if(onSegment(p, x, y)) return 0;
		bool c1 = sgn(x.y - p.y) > 0, c2 = sgn(y.y - p.y) > 0;
		if(c1 != c2) {
			ld xx = x.x + (y.x - x.x) * (p.y - x.y) / (y.y - x.y);
			if(sgn(xx - p.x) > 0) in ^= 1;
		}
	}
	return in ? 1 : -1;
}

// 求点集凸包，默认删除边界上的中间共线点
vector<Point> convexHull(vector<Point> a) {
	sort(a.begin(), a.end());
	vector<Point> p;
	for (auto x : a) if(p.empty() || x != p.back()) p.push_back(x);
	int n = p.size();
	if(n <= 1) return p;

	vector<Point> h;
	for (int i = 0; i < n; i++) {
		while(h.size() >= 2 && sgn(cross(h[h.size() - 2], h.back(), p[i])) <= 0) h.pop_back();
		h.push_back(p[i]);
	}

	int k = h.size();
	for (int i = n - 2; i >= 0; i--) {
		while((int)h.size() > k && sgn(cross(h[h.size() - 2], h.back(), p[i])) <= 0) h.pop_back();
		h.push_back(p[i]);
	}
	h.pop_back();
	return h;
}

// 旋转卡壳求凸包直径
ld convexDiameter(const vector<Point> &a) {
	int n = a.size();
	if(n <= 1) return 0;
	if(n == 2) return len2(a[0] - a[1]);

	int j = 1;
	ld ans = 0;
	for (int i = 0; i < n; i++) {
		int ni = (i + 1) % n;
		while(fabsl(cross(a[ni] - a[i], a[(j + 1) % n] - a[i])) > fabsl(cross(a[ni] - a[i], a[j] - a[i])) + eps) j = (j + 1) % n;
		ans = max(ans, len2(a[i] - a[j]));
		ans = max(ans, len2(a[ni] - a[j]));
	}
	return sqrtl(ans);
}
```

## 圆
```cpp
struct Circle {
	Point c;
	ld r;
	Circle(Point c_ = {}, ld r_ = 0) : c(c_), r(r_) {} // 构造圆
};

// 判断点与圆位置：-1 圆内，0 圆上，1 圆外
int pointCircleRelation(Point p, Circle c) {
	return sgn(dist(p, c.c) - c.r);
}

// 判断直线与圆位置：0 相离，1 相切，2 相交
int lineCircleRelation(Line l, Circle c) {
	ld d = distPointLine(c.c, l);
	if(sgn(d - c.r) > 0) return 0;
	if(sgn(d - c.r) == 0) return 1;
	return 2;
}

// 求直线与圆的交点
vector<Point> lineCircleIntersection(Line l, Circle c) {
	Point h = projection(c.c, l);
	ld d = dist(h, c.c);
	if(sgn(d - c.r) > 0) return {};
	if(sgn(d - c.r) == 0) return {h};

	ld x = sqrtl(max((ld)0, c.r * c.r - d * d));
	Point v = normalize(l.b - l.a);
	return {h - v * x, h + v * x};
}

// 判断两圆位置：-2 内含，-1 内切，0 相交，1 外切，2 外离，3 重合
int circleRelation(Circle a, Circle b) {
	ld d = dist(a.c, b.c);
	if(!sgn(d) && !sgn(a.r - b.r)) return 3;
	if(sgn(d - a.r - b.r) > 0) return 2;
	if(!sgn(d - a.r - b.r)) return 1;
	if(sgn(d - fabsl(a.r - b.r)) > 0) return 0;
	if(!sgn(d - fabsl(a.r - b.r))) return -1;
	return -2;
}

// 求两圆交点，重合时返回空数组
vector<Point> circleIntersection(Circle a, Circle b) {
	ld d = dist(a.c, b.c);
	if(!sgn(d)) return {};
	if(sgn(d - a.r - b.r) > 0) return {};
	if(sgn(d - fabsl(a.r - b.r)) < 0) return {};

	ld x = (d * d + a.r * a.r - b.r * b.r) / (2 * d);
	ld h2 = a.r * a.r - x * x;
	Point e = (b.c - a.c) / d;
	Point p = a.c + e * x;

	if(sgn(h2) == 0) return {p};

	ld h = sqrtl(max((ld)0, h2));
	Point v = rotate90(e) * h;
	return {p + v, p - v};
}
```
