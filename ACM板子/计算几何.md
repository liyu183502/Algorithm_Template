using ld = long double;
const ld eps = 1e-12;
const ld PI = acosl(-1);
int sgn(ld x) {
	if(fabsl(x) < eps) return 0;
	return x > 0 ? 1 : -1;
}

struct Point {
	ld x, y;
	Point(ld x_ = 0, ld y_ = 0) : x(x_), y(y_) {}
	Point operator + (const Point &p) const { return {x + p.x, y + p.y}; }
	Point operator - (const Point &p) const { return {x - p.x, y - p.y}; }
	Point operator * (ld k) const { return {x * k, y * k}; }
	Point operator / (ld k) const { return {x / k, y / k}; }
	Point operator - () const { return {-x, -y}; }
	bool operator == (const Point &p) const { return !sgn(x - p.x) && !sgn(y - p.y); }
	bool operator != (const Point &p) const { return !(*this == p); }
	bool operator < (const Point &p) const {
		if(x != p.x) return x < p.x;
		return y < p.y;
	}
};
ld dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
ld cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
ld cross(Point a, Point b, Point c) { return cross(b - a, c - a); } // >0逆时针，=0共线
ld len2(Point p) { return dot(p, p); } // 向量长度平方
ld len(Point p) { return sqrtl(len2(p)); } // 向量长度
ld dist(Point a, Point b) { return len(a - b); } // 两点距离


// 向量
Point normalize(Point p) { return p / len(p); } // 非0向量单位化
Point rotate(Point p, ld rad) { // 逆时针旋转 rad
	return {p.x * cosl(rad) - p.y * sinl(rad), p.x * sinl(rad) + p.y * cosl(rad)};
}
Point rotate90(Point p) { return {-p.y, p.x}; } // 逆时针旋转90度
ld angle(Point a, Point b) { // 两向量夹角
	ld x = dot(a, b) / len(a) / len(b);
	x = max((ld)-1, min((ld)1, x));
	return acosl(x);
}


// 点线位置关系
bool pointOnLineLeft(Point p, Point a, Point b) { // 点p在有向直线ab左侧
	return sgn(cross(b - a, p - a)) > 0;
}
bool onSegment(Point p, Point a, Point b) { // 点p在线段ab上
	return !sgn(cross(a, b, p)) && sgn(dot(p - a, p - b)) <= 0;
}


// 平行与相交
bool parallel(Point a, Point b, Point c, Point d) { // 直线ab与cd是否平行
	return !sgn(cross(b - a, d - c));
}
Point lineIntersection(Point a, Point b, Point c, Point d) { // 两条不平行直线交点
	Point v = b - a, w = d - c;
	ld t = cross(c - a, w) / cross(v, w);
	return a + v * t;
}
bool properIntersect(Point a, Point b, Point c, Point d) { // 两线段严格相交
	return sgn(cross(a, b, c)) * sgn(cross(a, b, d)) < 0 &&
		   sgn(cross(c, d, a)) * sgn(cross(c, d, b)) < 0;
}
bool segmentIntersect(Point a, Point b, Point c, Point d) { // 线段相交，包括端点和共线重叠
	int c1 = sgn(cross(a, b, c)), c2 = sgn(cross(a, b, d));
	int c3 = sgn(cross(c, d, a)), c4 = sgn(cross(c, d, b));
	if(c1 * c2 < 0 && c3 * c4 < 0) return true;
	if(c1 == 0 && onSegment(c, a, b)) return true;
	if(c2 == 0 && onSegment(d, a, b)) return true;
	if(c3 == 0 && onSegment(a, c, d)) return true;
	if(c4 == 0 && onSegment(b, c, d)) return true;
	return false;
}


// 距离相关
ld distPointLine(Point p, Point a, Point b) { // 点p到直线ab距离
	return fabsl(cross(b - a, p - a)) / len(b - a);
}
ld distPointSegment(Point p, Point a, Point b) { // 点p到线段ab距离
	if(a == b) return dist(p, a);
	if(sgn(dot(p - a, b - a)) < 0) return dist(p, a);
	if(sgn(dot(p - b, a - b)) < 0) return dist(p, b);
	return distPointLine(p, a, b);
}
ld distSegmentSegment(Point a, Point b, Point c, Point d) { // 两线段最短距离
	if(segmentIntersect(a, b, c, d)) return 0;
	return min({
		distPointSegment(a, c, d),
		distPointSegment(b, c, d),
		distPointSegment(c, a, b),
		distPointSegment(d, a, b)
	});
}


// 投影与对称
Point projection(Point p, Point a, Point b) { // 点p在直线ab上的投影
	Point v = b - a;
	return a + v * (dot(p - a, v) / dot(v, v));
}
Point reflection(Point p, Point a, Point b) { // 点p关于直线ab的对称点
	Point h = projection(p, a, b);
	return h * 2 - p;
}

// 精准版本极角序
int half(Point p) {
	if(sgn(p.y)) return p.y < 0;
	return p.x < 0;
}
bool polarCmp(Point a, Point b) { // 从x轴正方向开始逆时针排序
	int ha = half(a), hb = half(b);
	if(ha != hb) return ha < hb;
	int c = sgn(cross(a, b));
	if(c) return c > 0;
	return len2(a) < len2(b);
}

// 求多边形面积
ld polygonArea(const vector<Point> &a) {
	int n = a.size();
	ld res = 0;
	for(int i = 0; i < n; i++) res += cross(a[i], a[(i + 1) % n]);
	return fabsl(res) / 2;
}

// 求多边形周长
ld perimeter(const vector<Point> &a) {
	int n = a.size();
	ld res = 0;
	for(int i = 0; i < n; i++) res += dist(a[i], a[(i + 1) % n]);
	return res;
}

// 判断点与多边形位置：1内部，0边界，-1外部
int pointInPolygon(Point p, const vector<Point> &a) {
	int n = a.size();
	bool in = false;
	for(int i = 0; i < n; i++) {
		Point x = a[i], y = a[(i + 1) % n];
		if(onSegment(p, x, y)) return 0;
		bool c1 = sgn(x.y - p.y) > 0, c2 = sgn(y.y - p.y) > 0;
		if(c1 != c2) {
			ld xx = x.x + (y.x - x.x) * (p.y - x.y) / (y.y - x.y);
			if(sgn(xx - p.x) > 0) in ^= 1;
		}
	}
	return in ? 1 : -1;
}

// 求点集凸包，默认删除边界上的中间共线点
vector<Point> convexHull(vector<Point> a) {
	sort(a.begin(), a.end());
	vector<Point> p;
	for(auto x : a) if(p.empty() || x != p.back()) p.push_back(x);
	int n = p.size();
	if(n <= 1) return p;

	vector<Point> h;
	for(int i = 0; i < n; i++) {
		while(h.size() >= 2 && sgn(cross(h[h.size() - 2], h.back(), p[i])) <= 0) h.pop_back();
		h.push_back(p[i]);
	}

	int k = h.size();
	for(int i = n - 2; i >= 0; i--) {
		while((int)h.size() > k && sgn(cross(h[h.size() - 2], h.back(), p[i])) <= 0) h.pop_back();
		h.push_back(p[i]);
	}

	h.pop_back();
	return h;
}

// 旋转卡壳求凸包直径
ld convexDiameter(const vector<Point> &a) {
	int n = a.size();
	if(n <= 1) return 0;
	if(n == 2) return dist(a[0], a[1]);

	int j = 1;
	ld ans = 0;
	for(int i = 0; i < n; i++) {
		int ni = (i + 1) % n;
		while(fabsl(cross(a[ni] - a[i], a[(j + 1) % n] - a[i])) >
			  fabsl(cross(a[ni] - a[i], a[j] - a[i])) + eps) {
			j = (j + 1) % n;
		}
		ans = max(ans, len2(a[i] - a[j]));
		ans = max(ans, len2(a[ni] - a[j]));
	}
	return sqrtl(ans);
}

struct Circle {
	Point c;
	ld r;
	Circle(Point c_ = {}, ld r_ = 0) : c(c_), r(r_) {}
};

// 判断点与圆位置：-1圆内，0圆上，1圆外
int pointCircleRelation(Point p, Circle c) {
	return sgn(dist(p, c.c) - c.r);
}

// 判断直线ab与圆位置：0相离，1相切，2相交
int lineCircleRelation(Point a, Point b, Circle c) {
	ld d = distPointLine(c.c, a, b);
	if(sgn(d - c.r) > 0) return 0;
	if(sgn(d - c.r) == 0) return 1;
	return 2;
}

// 求直线ab与圆的交点
vector<Point> lineCircleIntersection(Point a, Point b, Circle c) {
	Point h = projection(c.c, a, b);
	ld d = dist(h, c.c);

	if(sgn(d - c.r) > 0) return {};
	if(sgn(d - c.r) == 0) return {h};

	ld x = sqrtl(max((ld)0, c.r * c.r - d * d));
	Point v = normalize(b - a);

	return {h - v * x, h + v * x};
}

// 判断两圆位置：-2内含，-1内切，0相交，1外切，2外离，3重合
int circleRelation(Circle a, Circle b) {
	ld d = dist(a.c, b.c);
	if(!sgn(d) && !sgn(a.r - b.r)) return 3;
	if(sgn(d - a.r - b.r) > 0) return 2;
	if(!sgn(d - a.r - b.r)) return 1;
	if(sgn(d - fabsl(a.r - b.r)) > 0) return 0;
	if(!sgn(d - fabsl(a.r - b.r))) return -1;
	return -2;
}

// 求两圆交点，重合时返回空数组
vector<Point> circleIntersection(Circle a, Circle b) {
	ld d = dist(a.c, b.c);

	if(!sgn(d)) return {};
	if(sgn(d - a.r - b.r) > 0) return {};
	if(sgn(d - fabsl(a.r - b.r)) < 0) return {};

	ld x = (d * d + a.r * a.r - b.r * b.r) / (2 * d);
	ld h2 = a.r * a.r - x * x;
	Point e = (b.c - a.c) / d;
	Point p = a.c + e * x;

	if(sgn(h2) == 0) return {p};

	ld h = sqrtl(max((ld)0, h2));
	Point v = rotate90(e) * h;

	return {p + v, p - v};
}
