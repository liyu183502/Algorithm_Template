struct Point {
    long double x, y;
    bool operator < (const Point &other) const {
        if (x != other.x) return x < other.x;
        return y < other.y;
    }
    bool operator == (const Point &other) const {
        return x == other.x && y == other.y;
    }
};

//叉乘
long double cross(Point a, Point b, Point c) {
    return (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);
}

//点积
long double dot(Point a, Point b) {
    return a.x * b.x + a.y * b.y;
}

//两点距离
long double dist(Point a, Point b) {
    long double X = a.x - b.x;
    long double Y = a.y - b.y;
    return sqrtl(X * X + Y * Y);
}

vector<Point> convex_hull(vector<Point> a) {
    sort(a.begin(), a.end());
    a.erase(unique(a.begin(), a.end()), a.end());
    int n = a.size();

    if (n <= 1) return a;

    vector<Point> lower, upper;
    for (int i = 0; i < n; i ++ ) {
        while (lower.size() >= 2 && cross(lower[lower.size() - 2], lower.back(), a[i]) <= 0) {
            lower.pop_back();
        }
        lower.push_back(a[i]);
    }

    for (int i = n - 1; i >= 0; i -- ) {
        while (upper.size() >= 2 && cross(upper[upper.size() - 2], upper.back(), a[i]) <= 0) {
            upper.pop_back();
        }
        upper.push_back(a[i]);
    }

    lower.pop_back();
    upper.pop_back();

    for (auto x : upper) {
        lower.push_back(x);
    }

    return lower;
} //lower 为逆时针凸包上面所有的点
