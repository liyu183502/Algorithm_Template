    auto ternary = [&](int l, int r) -> int {
        while(r - l > 3) {
            int m1 = l + (r - l) / 3;
            int m2 = r - (r - l) / 3;

            if(check(m1) < check(m2)) l = m1 + 1; // 最大值
            // if(check(m1) > check(m2)) l = m1 + 1; // 最小值

            else r = m2 - 1;
        }

        int ans = l;
        for (int i = l + 1; i <= r; i++)
            if(check(i) > check(ans)) ans = i; // 最大值
            // if(check(i) < check(ans)) ans = i; // 最小值

        return ans;
    };

    int x = ternary(l, r);
